56 // ---------------------------------------------------------------------------------------------------------
57 /// \enum NFunction

60 enum class NFunction : std::uint32_t
61 {
62 Input = 0b000, ///< Input pin
63 Output = 0b001, ///< Output pin
64 Alt_0 = 0b100, ///< Alternative function 0
65 Alt_1 = 0b101, ///< Alternative function 1
66 Alt_2 = 0b110, ///< Alternative function 2
67 Alt_3 = 0b111, ///< Alternative function 3
68 Alt_4 = 0b011, ///< Alternative function 4
69 Alt_5 = 0b010 ///< Alternative function 5

72 // ---------------------------------------------------------------------------------------------------------
73 /// \enum NState
74 /// \brief This enumeration defines states of a pin (only digital pins are supported).
75 // ---------------------------------------------------------------------------------------------------------
76 enum class NState : std::uint8_t
77 {
78 Low = 0, ///< Low voltage (0V)
79 High = 1 ///< High voltage (+5V)
80 };
81
82 // ---------------------------------------------------------------------------------------------------------
83 /// \enum NInterrupt_Type
84 /// \brief This enumeration defines different interrupt types that can be assigned to a pin.
85 // ---------------------------------------------------------------------------------------------------------
86 enum class NInterrupt_Type : std::uint8_t
87 {
88 Rising_Edge = 0, ///< Change from NState::Low to NState::High
89 Falling_Edge = 1, ///< Change from NState::High to NState::Low
90 Low = 2, ///< NState::Low detected
91 High = 3, ///< NState::High detected
92 Undefined = 4 ///< Undefined (helper enum record)
93 };

99 /// Alias for an array of interrupts (just to make the code less wordy)
100 using Interrupts_t = std::array<bool, Number_Of_Interrupt_Types>;

102 public:

183 // -------------------------------------------------------------------------------------------------------------
184 /// \enum NRegister
185 /// \brief Enumeration of different registers which are used to interact with the GPIO pins.
186 // -------------------------------------------------------------------------------------------------------------
187 enum class NRegister : std::uint32_t
188 {
189 GPFSEL0 = 0, ///< Function of GPIO pins 0-9 (3 bits per pin)
190 GPFSEL1, ///< Function of GPIO pins 10-19 (3 bits per pin)
191 GPFSEL2, ///< Function of GPIO pins 02-29 (3 bits per pin)
192 GPFSEL3, ///< Function of GPIO pins 30-39 (3 bits per pin)
193 GPFSEL4, ///< Function of GPIO pins 40-49 (3 bits per pin)
194 GPFSEL5, ///< Function of GPIO pins 50-23 (3 bits per pin)
195 Reserved_01, ///< Reserved
196 GPSET0, ///< Sets the value of a GPIO pin (pins 0-31)
197 GPSET1, ///< Sets the value of a GPIO pin (pins 32-53)
198 Reserved_02, ///< Reserved
199 GPCLR0, ///< Clears the value of a GPIO pin (pins 0-31)
200 GPCLR1, ///< Clears the value of a GPIO pin (pins 32-53)
201 Reserved_03, ///< Reserved
202 GPLEV0, ///< Actual value of the pin (pins 0-31)
203 GPLEV1, ///< Actual value of the pin (pins 32-53)
204 Reserved_04, ///< Reserved
205 GPEDS0, ///< GPIO Pin Event Detect Status 0 (pins 0-31)
206 GPEDS1, ///< GPIO Pin Event Detect Status 1 (pins 32-53)
207 Reserved_05, ///< Reserved
208 GPREN0, ///< Enables/disables the rising edge interrupt for pins 0-31
209 GPREN1, ///< Enables/disables the rising edge interrupt for pins 32-53
210 Reserved_06, ///< Reserved
211 GPFEN0, ///< Enables/disables the falling edge interrupt for pins 0-31
212 GPFEN1, ///< Enables/disables the falling edge interrupt for pins 32-53
213 Reserved_07, ///< Reserved
214 GPHEN0, ///< Enables/disables the high value interrupt for pins 0-31
215 GPHEN1, ///< Enables/disables the high value interrupt for pins 32-53
216 Reserved_08, ///< Reserved
217 GPLEN0, ///< Enables/disables the low value interrupt for pins 0-31
218 GPLEN1, ///< Enables/disables the low value interrupt for pins 32-53
219 Reserved_09, ///< Reserved
220 GPAREN0, ///< Enables/disables the asynchronous rising edge interrupt for pins 0-31
221 GPAREN1, ///< Enables/disables the asynchronous rising edge interrupt for pins 32-53
222 Reserved_10, ///< Reserved
223 GPAFEN0, ///< Enables/disables the asynchronous falling edge interrupt for pins 0-31
224 GPAFEN1, ///< Enables/disables the asynchronous falling edge interrupt for pins 32-53
225 Reserved_11, ///< Reserved
226 GPPUD, ///< Pull-up/down register controls
227 GPPUDCLK0, ///< Pull-up/down clock register 0
228 GPPUDCLK1, ///< Pull-up/down clock register 1
229 Reserved_12, ///< Reserved
230 Test, ///< Test
231 Count ///< Total number of registers (helper enum record)
232 };
