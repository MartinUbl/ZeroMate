26 /// \class CISA_Decoder
27 /// \brief This class represents an instruction decoder
28 ///
29 /// It is used in core.cpp to decode individual instructions as they are fetched from the memory.
30 // -----------------------------------------------------------------------------------------------------------------
31 class CISA_Decoder
32 {
33 private:
34 // -------------------------------------------------------------------------------------------------------------
35 /// \struct TInstruction_Lookup_Record
36 /// \brief This structure represents a single row in the instruction look-up table.
37 // -------------------------------------------------------------------------------------------------------------
38 struct TInstruction_Lookup_Record
39 {
40 std::uint32_t mask; ///< Instruction mask to mask out varying (unknown) bits
41 std::uint32_t expected; ///< Expected value of after the mask has been applied
42 CInstruction::NType type; ///< Corresponding instruction type
43 };
44
45 public:
46 // -------------------------------------------------------------------------------------------------------------
47 /// \brief Creates an instance of the class.
48 ///
49 /// Upon construction, it sorts out the look-up table from the most restrictive mask to the least
50 /// restrictive masks.
51 // -------------------------------------------------------------------------------------------------------------
52 CISA_Decoder() noexcept;
53
54 // -------------------------------------------------------------------------------------------------------------
55 /// \brief Looks up an instruction type in the instruction look-up table.
56 ///
57 /// The look-up time of this function is O(n) as it iterates row by row through the look-up table.
58 ///
59 /// \param instruction 32-bit value representing an encoded instruction
60 /// \return Corresponding instruction type
61 // -------------------------------------------------------------------------------------------------------------
62 [[nodiscard]] static CInstruction::NType Get_Instruction_Type(CInstruction instruction) noexcept;
63
64 private:
65 /// \brief Total number of rows in the look-up table (number of different instruction masks).
66 static constexpr std::size_t Instruction_Masks_Count = 25;
67
68 /// \brief Instruction look-up table made up of CISA_Decoder::TInstruction_Lookup_Record.
69 /// \note The table is not cost on purpose as it gets sorted in the constructor.
70 static std::array<TInstruction_Lookup_Record, Instruction_Masks_Count> s_instruction_lookup_table;
71 };
