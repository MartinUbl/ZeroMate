32 /// \class CARM_Timer
33 /// \brief This class represents the ARM timer used in BCM2835.
34 // -----------------------------------------------------------------------------------------------------------------
35 class CARM_Timer final : public IPeripheral, public ISystem_Clock_Listener
36 {
37 public:
38 // -------------------------------------------------------------------------------------------------------------
39 /// \enum NRegister
40 /// \brief Enumeration of different registers which are used to interact with the timer.
41 // -------------------------------------------------------------------------------------------------------------
42 enum class NRegister : std::uint32_t
43 {
44 Load = 0, ///< The value the timer starts off at (it it loaded into the value register)
45 Value, ///< The current value of the timer (counts down to 0)
46 Control, ///< Control register (see CARM_Timer::TControl_Register)
47 IRQ_Clear, ///< Used to clear a pending IRQ (by writing a 1 to it)
48 IRQ_Raw, ///< When the times reaches zero, this register is set to a 1 regardless of the settings
49 IRQ_Masked, ///< If the timer interrupt is enabled, this register is set to a 1 whenever the timer hits 0
50 Reload, ///< Copy of the load register
51 Pre_Divider, ///< TODO not used?
52 Free_Running, ///< Free running counter (how many times the timer has hit 0)
53 Count ///< Total number of register (helper enum record)
54 };
55
56 /// Collection of read-only registers (access control)
57 static const std::unordered_set<NRegister> s_read_only_registers;
58
59 /// Collection of write-only register (access control)
60 static const std::unordered_set<NRegister> s_write_only_registers;
61
62 // -------------------------------------------------------------------------------------------------------------
63 /// \enum NPrescal_Bits
64 /// \brief Different pre-scaler values that can be used to adjust the speed of the timer.
65 ///
66 /// The timer frequency is inherited from the main CPU clock.
67 // -------------------------------------------------------------------------------------------------------------
68 enum class NPrescal_Bits : std::uint32_t
69 {
70 Prescale_None = 0b00U, ///< No prescaler is present
71 Prescale_16 = 0b01U, ///< The CPU frequency is divided by 16
72 Prescale_256 = 0b10U, ///< The CPU frequency is divided by 256
73 Prescale_1 = 0b11U ///< The CPU frequency is divided by 1 (same as the CPU frequency)
74 };
75
76#pragma pack(push, 1) // so the compiler does not put any extra bits in between the struct members
77
78 // -------------------------------------------------------------------------------------------------------------
79 /// \struct TControl_Register
80 /// \brief Bit layout of the control register.
81 // -------------------------------------------------------------------------------------------------------------
82 struct TControl_Register
83 {
84 std::uint32_t Unused_0 : 1; ///< Unused
85 std::uint32_t Counter_32b : 1; ///< The timer either counts in 16-bit = 0 or 32-bit mode = 1
86 std::uint32_t Prescaler : 2; ///< Prescaler value (see CARM_Timer::NPrescal_Bits)
87 std::uint32_t Unused_1 : 1; ///< Unused
88 std::uint32_t Interrupt_Enabled : 1; ///< Enable/disable interrupts
89 std::uint32_t Unused_2 : 1; ///< Unused
90 std::uint32_t Timer_Enabled : 1; ///< Enable/disable the timer
91 std::uint32_t Halt_In_Debug_Break : 1; ///< // TODO have no clue as to what it does
92 std::uint32_t Free_Running : 1; ///< Enable free-running mode
93 std::uint32_t Unused_3 : 6; ///< Unused
94 std::uint32_t Free_Running_Prescaler : 8; ///< // TODO have no clue as to what it does
95 std::uint32_t Unused_4 : 8; ///< Unused
96 };
97
98#pragma pack(pop)

168 // -------------------------------------------------------------------------------------------------------------
169 /// \class CPrescaler
170 /// \brief This class represents a prescaler of the ARM timer.
171 ///
172 /// It works as wrapper counter around the number of cycles passed in whenever an instruction is executed.
173 // -------------------------------------------------------------------------------------------------------------
174 class CPrescaler final
175 {
176 public:
177 // ---------------------------------------------------------------------------------------------------------
178 /// \enum NPrescal_Values
179 /// \brief This represents the threshold the total number of CPU cycles must reach so the counter is increm.
180 // ---------------------------------------------------------------------------------------------------------
181 enum class NPrescal_Values : std::uint32_t
182 {
183 Prescale_16 = 16U, ///< 16 (the prescaler counts up to 16)
184 Prescale_256 = 256U, ///< 256 (the prescaler counts up to 256)
185 };
186
187 public:
188 // ---------------------------------------------------------------------------------------------------------
189 /// \brief Creates an instance of the class.
190 // ---------------------------------------------------------------------------------------------------------
191 CPrescaler();
192
193 // ---------------------------------------------------------------------------------------------------------
194 /// \brief Resets the prescaler.
195 ///
196 /// The default limit is set to NPrescal_Bits::Prescale_None.
197 // ---------------------------------------------------------------------------------------------------------
198 void Reset() noexcept;
199
200 // ---------------------------------------------------------------------------------------------------------
201 /// \brief Updates the prescaler.
202 ///
203 /// The prescaler adds this value to its own counter and checks whether the threshold has been reached.
204 ///
205 /// \param cycles_passed Number of CPU cycles it took to execute the last instruction
206 /// \return Updated (prescaled) number CPU cycles it took to execute the last instruction
207 // ---------------------------------------------------------------------------------------------------------
208 std::uint32_t operator()(std::uint32_t cycles_passed) noexcept;
209
210 // ---------------------------------------------------------------------------------------------------------
211 /// \brief Sets a new limit of the prescaler.
212 /// \param limit New prescaler value
213 // ---------------------------------------------------------------------------------------------------------
214 void Set_Limit(NPrescal_Bits limit);
215
216 private:
217 // ---------------------------------------------------------------------------------------------------------
218 /// \brief Updates the internal prescaler counter (helper function).
219 /// \param limit_value Threshold the internal counter counts up to
220 /// \return Number of prescaled cycles
221 // ---------------------------------------------------------------------------------------------------------
222 [[nodiscard]] std::uint32_t Update_Counter(std::uint32_t limit_value);
223
224 private:
225 NPrescal_Bits m_limit; ///< Prescaler settings
226 std::uint32_t m_counter; ///< Internal counter to keep track of the number of CPU cycles
227 };
228
229 private:
230 // -------------------------------------------------------------------------------------------------------------
231 /// \brief Clear a pending interrupt (helper function).
232 ///
233 /// When a 1 is written to this register, it notifies the interrupt controller, so it does not keep on firing
234 /// an IRQ exception. It also clears the following registers: IRQ_Clear, IRQ_Raw, IRQ_Masked.
235 // -------------------------------------------------------------------------------------------------------------
236 inline void Clear_Basic_IRQ();
237
238 // -------------------------------------------------------------------------------------------------------------
239 /// \brief This procedure is called whenever the timer reaches 0 (helper function).
240 ///
241 /// It sets the value register to the value stored in the reload register, and sets IRQ_Raw to 1.
242 /// If interrupts are enabled, it also stores a 1 into IRQ_Masked, and signalizes the interrupt controller.
243 /// If the free-running if set to a 1, it increments the Free_Running register.
244 ///
245 /// \param control_reg Reference to the timer control register
246 // -------------------------------------------------------------------------------------------------------------
247 inline void Timer_Has_Reached_Zero(const TControl_Register& control_reg);
248
249 private:
250 std::shared_ptr<CInterrupt_Controller> m_interrupt_controller; ///< Interrupt controller
251 std::array<std::uint32_t, Number_Of_Registers> m_regs; ///< Timer registers
252 CPrescaler m_prescaler; ///< Prescaler
253 utils::CLogging_System& m_logging_system; ///< Logging system
254 };
